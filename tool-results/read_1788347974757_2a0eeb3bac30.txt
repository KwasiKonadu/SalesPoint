   151→const PRODUCT_COLORS = [
   152→  'bg-emerald-500',
   153→  'bg-amber-500',
   154→  'bg-rose-500',
   155→  'bg-violet-500',
   156→  'bg-cyan-500',
   157→  'bg-orange-500',
   158→  'bg-teal-500',
   159→  'bg-pink-500',
   160→  'bg-lime-500',
   161→  'bg-fuchsia-500',
   162→];
   163→
   164→const getProductColor = (name: string): string => {
   165→  let hash = 0;
   166→  for (let i = 0; i < name.length; i++) {
   167→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
   168→  }
   169→  return PRODUCT_COLORS[Math.abs(hash) % PRODUCT_COLORS.length];
   170→};
   171→
   172→const getInitial = (name: string): string => {
   173→  return name.charAt(0).toUpperCase();
   174→};

   176→// ==================== Zod Schema ====================

   178→const productFormSchema = z.object({
   179→  name: z.string().min(1, 'Product name is required'),
   180→  sku: z.string().min(1, 'SKU is required'),
   181→  description: z.string().optional(),
   182→  categoryId: z.string().optional().nullable(),
   183→  productTypeId: z.string().optional().nullable(),
   184→  unitId: z.string().optional().nullable(),
   185→  costPrice: z.coerce.number().min(0, 'Must be 0 or greater'),
   186→  sellingPrice: z.coerce.number().min(0.01, 'Selling price is required'),
   187→  wholesalePrice: z.coerce.number().min(0).optional().nullable(),
   188→  imageUrl: z.string().optional().nullable(),
   189→  initialStock: z.coerce.number().int().min(0).optional(),
   190→  minStockLevel: z.coerce.number().int().min(0),
   191→  taxEnabled: z.boolean(),
   192→  taxRate: z.coerce.number().min(0).max(100),
   193→  isActive: z.boolean(),
   194→});
   195→
   196→type ProductFormValues = z.infer<typeof productFormSchema>;

   198→// ==================== Product Image Placeholder ====================

   200→function ProductImage({
   201→  name,
   202→  image,
   203→  size = 'sm',
   204→}: {
   205→  name: string;
   206→  image?: string | null;
   207→  size?: 'sm' | 'md' | 'lg';
   208→}) {
   209→  const sizeClasses = {
   210→    sm: 'h-9 w-9 text-xs',
   211→    md: 'h-12 w-12 text-sm',
   212→    lg: 'h-24 w-24 text-2xl',
   213→  };
   214→
   215→  if (image) {
   216→    return (
   217→      <img
   218→        src={image}
   219→        alt={name}
   220→        className={`${sizeClasses[size]} rounded-lg object-cover`}
   221→      />
   222→    );
   223→  }
   224→
   225→  return (
   226→    <div
   227→      className={`${sizeClasses[size]} ${getProductColor(name)} rounded-lg flex items-center justify-center text-white font-semibold shrink-0`}
   228→    >
   229→      {getInitial(name)}
   230→    </div>
   231→  );
   232→}

   234→// ==================== Stock Badge ====================
   235→
   236→function StockBadge({
   237→  quantity,
   238→  minStockLevel,
   239→  tracksStock = true,
   240→}: {
   241→  quantity: number;
   242→  minStockLevel: number;
   243→  tracksStock?: boolean;
   244→}) {
   245→  if (!tracksStock) {
   246→    return (
   247→      <Badge variant="outline" className="font-normal">
   248→        N/A
   249→      </Badge>
   250→    );
   251→  }
   252→
   253→  if (quantity === 0) {
   254→    return (
   255→      <Badge variant="destructive" className="font-normal tabular-nums">
   256→        {quantity}
   257→      </Badge>
   258→    );
   259→  }
   260→
   261→  if (quantity <= minStockLevel) {
   262→    return (
   263→      <span className="inline-flex items-center gap-1.5 rounded-md border border-amber-300 bg-amber-50 text-amber-700 px-2 py-0.5 text-xs font-medium tabular-nums">
   264→        <CircleDot className="size-2.5 fill-amber-500 text-amber-500" />
   265→        {quantity}
   266→      </span>
   267→    );
   268→  }
   269→
   270→  return (
   271→    <span className="inline-flex items-center gap-1.5 rounded-md border border-emerald-300 bg-emerald-50 text-emerald-700 px-2 py-0.5 text-xs font-medium tabular-nums">
   272→      <CircleDot className="size-2.5 fill-emerald-500 text-emerald-500" />
   273→      {quantity}
   274→    </span>
   275→  );
   276→}

   278→// ==================== Product Form Dialog ====================
   279→
   280→function ProductFormDialog({
   281→  open,
   282→  onOpenChange,
   283→  product,
   284→  productTypes,
   285→  categories,
   286→  units,
   287→  onSuccess,
   288→}: {
   289→  open: boolean;
   290→  onOpenChange: (open: boolean) => void;
   291→  product: ProductItem | null;
   292→  productTypes: ProductType[];
   293→  categories: Category[];
   294→  units: Unit[];
   295→  onSuccess: () => void;
   296→}) {
   297→  const { user } = useAuth();
   298→  const isEditing = !!product;
   299→  const [submitting, setSubmitting] = useState(false);
   300→
   301→  const selectedProductType = product?.productTypeId
   302→    ? productTypes.find((t) => t.id === product.productTypeId)
   303→    : null;
   304→
   305→  const {
   306→    register,
   307→    handleSubmit,
   308→    watch,
   309→    setValue,
   310→    reset,
   311→    formState: { errors },
   312→  } = useForm<ProductFormValues>({
   313→    defaultValues: {
   314→      name: product?.name || '',
   315→      sku: product?.sku || '',
   316→      description: product?.description || '',
   317→      categoryId: product?.categoryId || null,
   318→      productTypeId: product?.productTypeId || null,
   319→      unitId: product?.unitId || null,
   320→      costPrice: product?.costPrice || 0,
   321→      sellingPrice: product?.sellingPrice || 0,
   322→      wholesalePrice: product?.wholesalePrice || null,
   323→      imageUrl: product?.image || null,
   324→      initialStock: 0,
   325→      minStockLevel: product?.minStockLevel || 5,
   326→      taxEnabled: product?.taxEnabled || false,
   327→      taxRate: product?.taxRate || 0,
   328→      isActive: product?.isActive !== false,
   329→    },
   330→  });
   331→
   332→  const taxEnabled = watch('taxEnabled');
   333→  const isActive = watch('isActive');
   334→  const productTypeId = watch('productTypeId');
   335→  const tracksStock = isEditing
   336→    ? selectedProductType?.tracksStock ?? true
   337→    : productTypes.find((t) => t.id === productTypeId)?.tracksStock ?? true;
   338→
   339→  useEffect(() => {
   340→    if (open) {
   341→      reset({
   342→        name: product?.name || '',
   343→        sku: product?.sku || '',
   344→        description: product?.description || '',
   345→        categoryId: product?.categoryId || null,
   346→        productTypeId: product?.productTypeId || null,
   347→        unitId: product?.unitId || null,
   348→        costPrice: product?.costPrice || 0,
   349→        sellingPrice: product?.sellingPrice || 0,
   350→        wholesalePrice: product?.wholesalePrice || null,
   351→        imageUrl: product?.image || null,
   352→        initialStock: 0,
   353→        minStockLevel: product?.minStockLevel || 5,
   354→        taxEnabled: product?.taxEnabled || false,
   355→        taxRate: product?.taxRate || 0,
   356→        isActive: product?.isActive !== false,
   357→      });
   358→    }
   359→  }, [open, product, reset]);
   360→
   361→  const onSubmit = async (data: ProductFormValues) => {
   362→    if (!user) return;
   363→    setSubmitting(true);
   364→
   365→    try {
   366→      const body: Record<string, unknown> = {
   367→        name: data.name,
   368→        sku: data.sku || undefined,
   369→        description: data.description || undefined,
   370→        categoryId: data.categoryId || undefined,
   371→        productTypeId: data.productTypeId || undefined,
   372→        unitId: data.unitId || undefined,
   373→        costPrice: data.costPrice,
   374→        sellingPrice: data.sellingPrice,
   375→        wholesalePrice: data.wholesalePrice || undefined,
   376→        taxEnabled: data.taxEnabled,
   377→        taxRate: data.taxEnabled ? data.taxRate : 0,
   378→        minStockLevel: data.minStockLevel,
   379→        image: data.imageUrl || undefined,
   380→      };
   381→
   382→      if (!isEditing && tracksStock) {
   383→        body.initialStock = data.initialStock;
   384→      }
   385→
   386→      if (isEditing) {
   387→        body.isActive = data.isActive;
   388→      }
   389→
   390→      const url = isEditing
   391→        ? `/api/products/${product.id}`
   392→        : '/api/products';
   393→      const method = isEditing ? 'PUT' : 'POST';
   394→
   395→      const res = await fetch(url, {
   396→        method,
   397→        headers: {
   398→          'Content-Type': 'application/json',
   399→          'x-user-id': user.id,
   400→        },
   401→        body: JSON.stringify(body),
   402→      });
   403→
   404→      if (!res.ok) {
   405→        const err = await res.json();
   406→        throw new Error(err.error || 'Failed to save product');
   407→      }
   408→
   409→      toast.success(
   410→        isEditing
   411→          ? 'Product updated successfully'
   412→          : 'Product created successfully'
   413→      );
   414→      onOpenChange(false);
   415→      onSuccess();
   416→    } catch (err) {
   417→      toast.error(
   418→        err instanceof Error ? err.message : 'Failed to save product'
   419→      );
   420→    } finally {
   421→      setSubmitting(false);
   422→    }
   423→  };
   424→
   425→  return (
   426→    <Dialog open={open} onOpenChange={onOpenChange}>
   427→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   428→        <DialogHeader>
   429→          <DialogTitle>
   430→            {isEditing ? 'Edit Product' : 'Add New Product'}
   431→          </DialogTitle>
   432→          <DialogDescription>
   433→            {isEditing
   434→              ? 'Update the product details below.'
   435→              : 'Fill in the details to create a new product.'}
   436→          </DialogDescription>
   437→        </DialogHeader>
   438→
   439→        <form onSubmit={handleSubmit(onSubmit)} className="space-y-6">
   440→          {/* Basic Information */}
   441→          <div className="space-y-4">
   442→            <h4 className="text-sm font-medium text-muted-foreground">Basic Information</h4>
   443→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   444→              <div className="space-y-2">
   445→                <Label htmlFor="name">
   446→                  Product Name <span className="text-destructive">*</span>
   447→                </Label>
   448→                <Input
   449→                  id="name"
   450→                  placeholder="e.g., Coca-Cola Can"
   451→                  {...register('name')}
   452→                  aria-invalid={!!errors.name}
   453→                />
   454→                {errors.name && (
   455→                  <p className="text-xs text-destructive">{errors.name.message}</p>
   456→                )}
   457→              </div>
   458→
   459→              <div className="space-y-2">
   460→                <Label htmlFor="sku">
   461→                  SKU <span className="text-destructive">*</span>
   462→                </Label>
   463→                <Input
   464→                  id="sku"
   465→                  placeholder="Leave empty to auto-generate"
   466→                  {...register('sku')}
   467→                  aria-invalid={!!errors.sku}
   468→                />
   469→                {errors.sku && (
   470→                  <p className="text-xs text-destructive">{errors.sku.message}</p>
   471→                )}
   472→              </div>
   473→            </div>
   474→
   475→            <div className="space-y-2">
   476→              <Label htmlFor="description">Description</Label>
   477→              <Textarea
   478→                id="description"
   479→                placeholder="Optional product description..."
   480→                rows={2}
   481→                {...register('description')}
   482→              />
   483→            </div>
   484→          </div>
   485→
   486→          <Separator />
   487→
   488→          {/* Classification */}
   489→          <div className="space-y-4">
   490→            <h4 className="text-sm font-medium text-muted-foreground">Classification</h4>
   491→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   492→              <div className="space-y-2">
   493→                <Label>Category</Label>
   494→                <Select
   495→                  value={watch('categoryId') || '_none'}
   496→                  onValueChange={(val) =>
   497→                    setValue(
   498→                      'categoryId',
   499→                      val === '_none' ? null : val
   500→                    )
   501→                  }
   502→                >
   503→                  <SelectTrigger className="w-full">
   504→                    <SelectValue placeholder="Select category" />
   505→                  </SelectTrigger>
   506→                  <SelectContent>
   507→                    <SelectItem value="_none">None</SelectItem>
   508→                    {categories.map((cat) => (
   509→                      <SelectItem key={cat.id} value={cat.id}>
   510→                        {cat.name}
   511→                      </SelectItem>
   512→                    ))}
   513→                  </SelectContent>
   514→                </Select>
   515→              </div>
   516→
   517→              <div className="space-y-2">
   518→                <Label>Product Type</Label>
   519→                <Select
   520→                  value={watch('productTypeId') || '_none'}
   521→                  onValueChange={(val) =>
   522→                    setValue(
   523→                      'productTypeId',
   524→                      val === '_none' ? null : val
   525→                    )
   526→                  }
   527→                >
   528→                  <SelectTrigger className="w-full">
   529→                    <SelectValue placeholder="Select type" />
   530→                  </SelectTrigger>
   531→                  <SelectContent>
   532→                    <SelectItem value="_none">None</SelectItem>

   534→                      <SelectItem key={type.id} value={type.id}>
   535→                        {type.name}

   537→                    ))}
   538→                  </SelectContent>

   542→              <div className="space-y-2">

   570→          <div className="space-y-4">
   571→            <h4 className="text-sm font-medium text-muted-foreground">Pricing</h4>

   755→  onOpenChange,
   756→  category,
   757→  onSuccess,

   759→  open: boolean;
   760→  onOpenChange: (open: boolean) => void;

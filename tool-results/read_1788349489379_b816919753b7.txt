   181→function formatDateForInput(d: Date): string {
   182→  const year = d.getFullYear();
   183→  const month = String(d.getMonth() + 1).padStart(2, '0');
   184→  const day = String(d.getDate()).padStart(2, '0');
   185→  return `${year}-${month}-${day}`;
   186→}

   188→function getPeriodDates(period: PeriodKey, customStart?: string, customEnd?: string) {
   189→  const now = new Date();
   190→  let startDate: Date;
   191→  let endDate: Date;
   192→
   193→  switch (period) {
   194→    case 'today':
   195→      startDate = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   196→      endDate = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59, 999);
   197→      break;
   198→    case 'this_week': {
   199→      const day = now.getDay();
   200→      const diff = day === 0 ? 6 : day - 1;
   201→      startDate = new Date(now);
   202→      startDate.setDate(now.getDate() - diff);
   203→      startDate.setHours(0, 0, 0, 0);
   204→      endDate = new Date(now);
   205→      endDate.setHours(23, 59, 59, 999);
   206→      break;
   207→    }
   208→    case 'this_month':
   209→      startDate = new Date(now.getFullYear(), now.getMonth(), 1);
   210→      endDate = new Date(now.getFullYear(), now.getMonth() + 1, 0, 23, 59, 59, 999);
   211→      break;
   212→    case 'last_month':
   213→      startDate = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   214→      endDate = new Date(now.getFullYear(), now.getMonth(), 0, 23, 59, 59, 999);
   215→      break;
   216→    case 'this_year':
   217→      startDate = new Date(now.getFullYear(), 0, 1);
   218→      endDate = new Date(now.getFullYear(), 11, 31, 23, 59, 59, 999);
   219→      break;
   220→    case 'custom':
   221→      startDate = new Date(customStart || now);
   222→      endDate = new Date(customEnd || now);
   223→      endDate.setHours(23, 59, 59, 999);
   224→      break;
   225→  }
   226→
   227→  return {
   228→    startDate: formatDateForInput(startDate),
   229→    endDate: formatDateForInput(endDate),
   230→  };
   231→}
   232→
   233→function getPreviousPeriodDates(period: PeriodKey, currentStart: string, currentEnd: string) {
   234→  const start = new Date(currentStart + 'T00:00:00');
   235→  const end = new Date(currentEnd + 'T00:00:00');
   236→  const diffMs = end.getTime() - start.getTime();
   237→  const prevEnd = new Date(start.getTime() - 1);
   238→  const prevStart = new Date(prevEnd.getTime() - diffMs);
   239→  return {
   240→    startDate: formatDateForInput(prevStart),
   241→    endDate: formatDateForInput(prevEnd),
   242→  };
   243→}
   244→
   245→function calcChange(current: number, previous: number): number {
   246→  if (previous === 0) return current > 0 ? 100 : 0;
   247→  return ((current - previous) / previous) * 100;
   248→}

   129→const getProductColor = (name: string): string => {
   130→  let hash = 0;
   131→  for (let i = 0; i < name.length; i++) {
   132→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
   133→  }
   134→  return PRODUCT_COLORS[Math.abs(hash) % PRODUCT_COLORS.length];

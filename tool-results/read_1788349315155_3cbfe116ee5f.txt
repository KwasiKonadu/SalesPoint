    92→// --- Helpers ---

    94→function formatCurrency(value: number): string {
    95→  return `\u20A5${value.toLocaleString('en-GH', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
    96→}
    97→
    98→function formatDate(dateStr: string): string {
    99→  return new Date(dateStr).toLocaleDateString('en-US', {
   100→    month: 'short',
   101→    day: 'numeric',
   102→    year: 'numeric',
   103→  });
   104→}
   105→
   106→function formatChartDate(dateStr: string): string {
   107→  const d = new Date(dateStr);
   108→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   109→}

   113→function ChangeIndicator({ change, label }: { change: number; label: string }) {
   114→  const isPositive = change >= 0;
   115→  return (
   116→    <div className="flex items-center gap-1 text-xs">
   117→      {isPositive ? (
   118→        <ArrowUpRight className="w-3.5 h-3.5 text-emerald-600" />
   119→      ) : (
   120→        <ArrowDownRight className="w-3.5 h-3.5 text-red-500" />
   121→      )}
   122→      <span className={isPositive ? 'text-emerald-600 font-medium' : 'text-red-500 font-medium'}>
   123→        {Math.abs(change).toFixed(1)}%
   124→      </span>
   125→      <span className="text-muted-foreground">vs last month</span>
   126→    </div>
   127→  );
   128→}

   146→
   147→function KPICard({
   148→  icon: Icon,
   149→  label,
   150→  value,
   151→  change,
   152→  format = 'number',
   153→  iconBg,
   154→}: {
   155→  icon: React.ElementType;
   156→  label: string;
   157→  value: number;
   158→  change: number;
   159→  format?: 'currency' | 'number' | 'decimal';
   160→  iconBg: string;
   161→}) {
   162→  const displayValue =
   163→    format === 'currency'
   164→      ? formatCurrency(value)
   165→      : format === 'decimal'
   166→        ? value.toLocaleString('en-GH', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
   167→        : value.toLocaleString();
   168→
   169→  return (
   170→    <Card className="shadow-sm hover:shadow-md transition-shadow">
   171→      <CardHeader className="pb-2">
   172→        <div className="flex items-center justify-between">
   173→          <p className="text-sm text-muted-foreground font-medium">{label}</p>
   174→          <div className={`h-9 w-9 rounded-lg flex items-center justify-center ${iconBg}`}>
   175→            <Icon className="w-4.5 h-4.5" />
   176→          </div>
   177→        </div>
   178→      </CardHeader>
   179→      <CardContent className="pt-0">
   180→        <p className="text-2xl font-bold tracking-tight">{displayValue}</p>
   181→        <div className="mt-1">
   182→          <ChangeIndicator change={change} label={label} />
   183→        </div>
   184→      </CardContent>
   185→    </Card>
   186→  );
   187→}

   189→// --- Custom Chart Tooltip ---
   190→
   191→function SalesChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number }>; label?: string }) {
   192→  if (!active || !payload?.length) return null;
   193→  return (
   194→    <div className="bg-popover border rounded-lg px-3 py-2 shadow-md">
   195→      <p className="text-xs text-muted-foreground mb-1">{label}</p>
   196→      <p className="text-sm font-semibold">{formatCurrency(payload[0].value)}</p>
   197→    </div>
   198→  );
   199→}
   200→
   201→function PieTooltip({ active, payload }: { active?: boolean; payload?: Array<{ name: string; value: number; payload: { percentage: number } }> }) {
   202→  if (!active || !payload?.length) return null;
   203→  const item = payload[0];
   204→  return (
   205→    <div className="bg-popover border rounded-lg px-3 py-2 shadow-md">
   206→      <p className="text-sm font-semibold">{item.name}</p>
   207→      <p className="text-xs text-muted-foreground">
   208→        {formatCurrency(item.value)} ({item.payload.percentage.toFixed(1)}%)
   209→      </p>
   210→    </div>
   211→  );
   212→}
